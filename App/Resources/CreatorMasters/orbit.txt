function setupScene(api){const T=api.THREE,scene=api.scene;scene.background=new T.Color('#01030a');const core=new T.Mesh(new T.IcosahedronGeometry(1.05,1),new T.MeshPhysicalMaterial({color:api.parameters.coreColor,metalness:.88,roughness:.19,clearcoat:1}));scene.add(core);
 const cage=new T.LineSegments(new T.EdgesGeometry(new T.IcosahedronGeometry(1.14,1)),new T.LineBasicMaterial({color:api.parameters.ringColor,transparent:true,opacity:.8}));scene.add(cage);
 const group=new T.Group();scene.add(group);const particles=new T.InstancedMesh(new T.IcosahedronGeometry(.018,0),new T.MeshStandardMaterial({color:0xffffff,emissive:api.parameters.ringColor,emissiveIntensity:2.4,roughness:.2,metalness:.1}),2048);const dummy=new T.Object3D(),color=new T.Color();
 let seed=19;const random=()=>{seed=(seed*1664525+1013904223)>>>0;return seed/4294967296};
 for(let i=0;i<2048;i++){const ring=i%4,angle=random()*Math.PI*2,r=2.15+ring*.42+(random()-.5)*.25;dummy.position.set(Math.cos(angle)*r,(random()-.5)*.14,Math.sin(angle)*r);dummy.rotation.set(angle,random()*6,angle);dummy.scale.setScalar(.45+Math.pow(random(),3)*2.6);dummy.updateMatrix();particles.setMatrixAt(i,dummy.matrix);color.setHSL(.49+random()*.24,.7,.55);particles.setColorAt(i,color);}group.add(particles);
 const rings=[];for(let i=0;i<3;i++){const mesh=new T.Mesh(new T.TorusGeometry(2.25+i*.6,.008,4,180),new T.MeshBasicMaterial({color:api.parameters.ringColor,transparent:true,opacity:.5}));mesh.rotation.x=Math.PI/2+i*.14;group.add(mesh);rings.push(mesh);}group.rotation.x=.52;
 const light=new T.PointLight(0x74cfff,65,16);light.position.set(3,3,4);scene.add(light);const back=new T.DirectionalLight(0xaf69ff,2);back.position.set(-4,-1,1);scene.add(back);return{core,cage,group,particles,rings,light};}
function updateScene(api,dt){const p=api.parameters,s=api.state,t=api.time*p.speed;api.basePosition.z=Math.max(10,9/(api.camera.aspect||1));s.core.rotation.set(t*.11,t*.2,Math.sin(t*.17)*.12);s.cage.rotation.copy(s.core.rotation);s.cage.scale.setScalar(1+Math.sin(t*1.1)*p.pulse*.035);
 s.group.rotation.y=t*.08;s.group.rotation.z=(api.cursor.x-.5)*p.tilt;s.group.rotation.x=.5+(api.cursor.y-.5)*p.tilt;s.particles.count=Math.floor(p.density);s.particles.material.emissive.set(p.ringColor);s.particles.material.emissiveIntensity=p.radiance;s.core.material.color.set(p.coreColor);s.cage.material.color.set(p.ringColor);for(const ring of s.rings)ring.material.color.set(p.ringColor);}
