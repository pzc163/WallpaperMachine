float hash21(vec2 p){p=fract(p*vec2(123.34,456.21));p+=dot(p,p+45.32);return fract(p.x*p.y);}
float noise(vec2 p){vec2 i=floor(p),f=fract(p);f=f*f*(3.-2.*f);return mix(mix(hash21(i),hash21(i+vec2(1,0)),f.x),mix(hash21(i+vec2(0,1)),hash21(i+1.),f.x),f.y);}
float fbm(vec2 p){float v=0.,a=.5;for(int k=0;k<4;k++){v+=a*noise(p);p=mat2(1.6,1.2,-1.2,1.6)*p+2.7;a*=.5;}return v;}
// Layered moving menisci; derivatives refract the actual selected media.
float drops(vec2 uv,float t){float z=0.;for(int layer=0;layer<3;layer++){float l=float(layer);vec2 g=uv*vec2(24.+l*9.,9.+l*5.);g.y+=t*(.35+l*.18);vec2 cell=floor(g);float h=hash21(cell+l*11.);vec2 q=fract(g)-.5;q+=vec2(.3*sin(cell.y*2.3+h*8.),(hash21(cell+17.)-.5)*.3);float head=exp(-dot(q*vec2(5.+h*4.,2.5+h*2.),q*vec2(5.+h*4.,2.5+h*2.)));float trail=exp(-q.x*q.x*160.)*smoothstep(.45,-.1,q.y)*smoothstep(-.5,.0,q.y);z+=(head+.2*trail)*step(.3,h)/3.;}return z;}
void mainImage(out vec4 c,in vec2 xy){vec2 uv=xy/iResolution.xy;float t=iTime*p_speed;float eps=1./iResolution.y;float d=drops(uv,t);vec2 grad=vec2(drops(uv+vec2(eps,0),t)-drops(uv-vec2(eps,0),t),drops(uv+vec2(0,eps),t)-drops(uv-vec2(0,eps),t))/(2.*eps);
vec2 cursor=vec2(iCursor.x,1.-iCursor.y);float clear=1.-p_wipe*iActive*exp(-dot(uv-cursor,uv-cursor)*30.);vec2 q=uv+grad*.00045*p_refraction*clear;vec3 col=wmBase(q);vec3 mist=(wmBase(q+vec2(.003,0))+wmBase(q-vec2(.003,0))+wmBase(q+vec2(0,.003))+wmBase(q-vec2(0,.003)))*.25;col=mix(col,mist,p_fog*(1.-d*.5));col+=vec3(.07,.11,.15)*d*p_highlight;col*=1.-p_vignette*pow(length((uv-.5)*vec2(1.3,1.)),1.7);c=vec4(col,1);}
