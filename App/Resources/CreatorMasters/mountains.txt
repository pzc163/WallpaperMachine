function setupScene(api){const T=api.THREE,scene=api.scene;scene.background=new T.Color('#060c23');scene.fog=new T.FogExp2('#142144',.035);
 const geometry=new T.PlaneGeometry(28,22,52,44);geometry.rotateX(-Math.PI/2);const positions=geometry.attributes.position,base=[],colors=[];const vertexColor=new T.Color();
 for(let i=0;i<positions.count;i++){const x=positions.getX(i),z=positions.getZ(i);const ridge=1.3*Math.exp(-Math.pow((z+3)/6,2));const peaks=3.8*Math.exp(-((x+5)*(x+5)/15+(z+4)*(z+4)/22))+4.2*Math.exp(-((x-3)*(x-3)/20+(z+7)*(z+7)/14));const jagged=Math.abs(Math.sin(x*.83+z*.47))*Math.abs(Math.cos(z*.67-x*.39))*.9+Math.sin(x*1.9+z*.9)*.17;const height=ridge+peaks+jagged*Math.exp(-Math.pow((z+3)/9,2))-3.4;positions.setY(i,height);base.push(height);vertexColor.setHSL(.58-.025*height,.45,.17+Math.max(0,height+3.4)*.055);colors.push(vertexColor.r,vertexColor.g,vertexColor.b);}geometry.setAttribute('color',new T.Float32BufferAttribute(colors,3));geometry.computeVertexNormals();
 const mat=new T.MeshStandardMaterial({color:api.parameters.terrain,roughness:.8,metalness:.28,flatShading:true,vertexColors:true,envMapIntensity:.16});const mesh=new T.Mesh(geometry,mat);scene.add(mesh);
 const key=new T.DirectionalLight(0x6ce6d1,2.5);key.position.set(-4,8,-6);scene.add(key);const rim=new T.DirectionalLight(0xd68cee,1.8);rim.position.set(5,3,5);scene.add(rim);scene.add(new T.HemisphereLight(0x426caa,0x030612,.35));
 const curtainGeo=new T.PlaneGeometry(30,7,1,1);const uniforms={time:{value:0},tint:{value:new T.Color(api.parameters.aurora)},strength:{value:1}};
 const curtainMat=new T.ShaderMaterial({uniforms,transparent:true,depthWrite:false,side:T.DoubleSide,blending:T.AdditiveBlending,vertexShader:'varying vec2 uvw;void main(){uvw=uv;gl_Position=projectionMatrix*modelViewMatrix*vec4(position,1.);}',fragmentShader:'precision highp float;varying vec2 uvw;uniform float time,strength;uniform vec3 tint;float n(float x){return fract(sin(x*127.1)*43758.5453);}float noise(float x){float f=fract(x);f=f*f*(3.-2.*f);return mix(n(floor(x)),n(floor(x)+1.),f);}void main(){float fold=noise(uvw.x*9.+time*.05)*.12+sin(uvw.x*12.+time*.1)*.06;float base=.18+fold;float height=uvw.y-base;float veil=exp(-max(height,0.)*5.)*smoothstep(-.025,.04,height);float rays=.35+.65*noise(uvw.x*85.+noise(uvw.x*17.)*5.+time*.04);float alpha=veil*rays*(1.-smoothstep(.6,.9,uvw.y))*smoothstep(0.,.08,uvw.x)*(1.-smoothstep(.92,1.,uvw.x));vec3 color=mix(tint,vec3(.38,.12,.62),smoothstep(.05,.6,height));gl_FragColor=vec4(color*alpha*strength,alpha*.45);}'});
 const curtain=new T.Mesh(curtainGeo,curtainMat);curtain.position.set(0,4.5,-12);scene.add(curtain);api.target.set(0,-.4,-4);return{mesh,base,key,rim,uniforms};}
function updateScene(api,dt){const p=api.parameters,t=api.time*p.speed,s=api.state;api.basePosition.set(0,4.2,Math.max(16,13/(api.camera.aspect||1)));const position=s.mesh.geometry.attributes.position;
 for(let i=0;i<position.count;i++){const x=position.getX(i),z=position.getZ(i);position.setY(i,s.base[i]*p.height+Math.sin(t*.45+x*.32+z*.2)*p.breathing);}position.needsUpdate=true;
 s.mesh.material.color.set(p.terrain);s.uniforms.time.value=t;s.uniforms.tint.value.set(p.aurora);s.uniforms.strength.value=p.auroraGlow;api.scene.fog.density=p.fog;s.key.color.set(p.aurora);}
