float hash21(vec2 p){p=fract(p*vec2(123.34,456.21));p+=dot(p,p+45.32);return fract(p.x*p.y);}
float noise(vec2 p){vec2 i=floor(p),f=fract(p);f=f*f*(3.-2.*f);return mix(mix(hash21(i),hash21(i+vec2(1,0)),f.x),mix(hash21(i+vec2(0,1)),hash21(i+1.),f.x),f.y);}
float fbm(vec2 p){float v=0.,a=.5;for(int k=0;k<4;k++){v+=a*noise(p);p=mat2(1.6,1.2,-1.2,1.6)*p+2.7;a*=.5;}return v;}
// Luminance-assisted 2.5D layers: not a neural depth reconstruction.
void mainImage(out vec4 c,in vec2 xy){vec2 uv=xy/iResolution.xy;vec2 offset=vec2(iCursor.x-.5,.5-iCursor.y)*p_parallax;float depth=clamp((1.-uv.y)*.7+wmLuma(uv)*.3,0.,1.);vec3 back=wmBase(uv+offset*.2);vec3 middle=wmBase(uv+offset*(.35+depth*.45));vec3 front=wmBase(uv+offset);float nearMask=smoothstep(.55,.95,depth);vec3 col=mix(back,middle,smoothstep(.1,.75,depth));col=mix(col,front,nearMask*.55);float fog=(1.-smoothstep(.35,.8,uv.y))*fbm(uv*vec2(3,2)+vec2(iTime*.015*p_speed,0));col+=p_haze*vec3(.17,.13,.18)*fog;float sunlight=exp(-dot(uv-vec2(.65,.68),uv-vec2(.65,.68))*9.);col+=p_warmth*sunlight*vec3(.1,.045,.018);col*=1.-p_vignette*dot(uv-.5,uv-.5);c=vec4(col,1);}
