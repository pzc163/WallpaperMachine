function setupScene(api){const T=api.THREE;api.scene.background=new T.Color('#080b16');api.scene.fog=new T.Fog('#080b16',16,32);
 const shape=new T.Shape(),r=.17,w=.44;shape.moveTo(-w+r,-w);shape.lineTo(w-r,-w);shape.quadraticCurveTo(w,-w,w,-w+r);shape.lineTo(w,w-r);shape.quadraticCurveTo(w,w,w-r,w);shape.lineTo(-w+r,w);shape.quadraticCurveTo(-w,w,-w,w-r);shape.lineTo(-w,-w+r);shape.quadraticCurveTo(-w,-w,-w+r,-w);
 const geometry=new T.ExtrudeGeometry(shape,{depth:.36,bevelEnabled:true,bevelSegments:4,steps:1,bevelSize:.1,bevelThickness:.1,curveSegments:12});geometry.center();
 const group=new T.Group(),cells=[];for(let y=0;y<3;y++)for(let x=0;x<5;x++){const material=new T.MeshPhysicalMaterial({color:api.parameters.tint,metalness:0,roughness:.18,transmission:.93,thickness:2.5,ior:1.48,dispersion:.45,clearcoat:1,clearcoatRoughness:.12,attenuationDistance:5,attenuationColor:api.parameters.tint});const mesh=new T.Mesh(geometry,material);mesh.position.set((x-2)*1.35,(y-1)*1.35,0);mesh.rotation.set(.12+(y-1)*.08,(x-2)*.08,.12);group.add(mesh);cells.push({mesh,x,y});}api.scene.add(group);
 const floor=new T.Mesh(new T.PlaneGeometry(40,40),new T.MeshStandardMaterial({color:0x03040a,metalness:.12,roughness:.55,envMapIntensity:.12}));floor.rotation.x=-Math.PI/2;floor.position.y=-2.45;api.scene.add(floor);
 const backdrop=new T.Group();for(let i=0;i<11;i++){const strip=new T.Mesh(new T.BoxGeometry(.055,6,.04),new T.MeshBasicMaterial({color:i%2?0x6c4aac:0x97bddd}));strip.position.set((i-5)*.62,0,-1.2);strip.rotation.z=-.35;backdrop.add(strip);}api.scene.add(backdrop);
 const key=new T.PointLight(0x9c8aff,95,24,2);key.position.set(3,2.8,3);api.scene.add(key);const fill=new T.DirectionalLight(0xb7e8ff,2.2);fill.position.set(-4,4,2);api.scene.add(fill);
 const halo=new T.Mesh(new T.TorusGeometry(3.9,.012,6,128),new T.MeshBasicMaterial({color:0x7263b8}));halo.position.z=-1.8;group.add(halo);return{group,cells,key,halo};}
function updateScene(api,dt){const p=api.parameters,t=api.time*p.speed,s=api.state;api.basePosition.z=Math.max(9,7/(api.camera.aspect||1));s.group.rotation.y=Math.sin(t*.13)*.12;s.group.rotation.z=Math.sin(t*.08)*.025;
 for(const {mesh,x,y} of s.cells){mesh.position.z=Math.sin(t*.65+x*.7+y*.55)*p.floatAmount;mesh.rotation.y=(x-2)*.08+Math.sin(t*.3+x)*.09;mesh.material.color.set(p.tint);mesh.material.attenuationColor.set(p.tint);mesh.material.roughness=p.frost;mesh.material.transmission=p.transmission;mesh.material.dispersion=p.dispersion;}
 s.key.position.set((api.cursor.x-.5)*7,2+(1-api.cursor.y)*2,3);s.key.color.set(p.lightColor);s.key.intensity=p.lightPower;}
