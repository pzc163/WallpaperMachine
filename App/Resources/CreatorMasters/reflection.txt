float hash21(vec2 p){p=fract(p*vec2(123.34,456.21));p+=dot(p,p+45.32);return fract(p.x*p.y);}
float noise(vec2 p){vec2 i=floor(p),f=fract(p);f=f*f*(3.-2.*f);return mix(mix(hash21(i),hash21(i+vec2(1,0)),f.x),mix(hash21(i+vec2(0,1)),hash21(i+1.),f.x),f.y);}
float fbm(vec2 p){float v=0.,a=.5;for(int k=0;k<4;k++){v+=a*noise(p);p=mat2(1.6,1.2,-1.2,1.6)*p+2.7;a*=.5;}return v;}
void mainImage(out vec4 c,in vec2 xy){vec2 uv=xy/iResolution.xy;float water=smoothstep(p_horizon+.035,p_horizon-.025,uv.y);float distance=max(.02,p_horizon-uv.y);float t=iTime*p_speed;float wave=sin(uv.x*52.+t*1.3+fbm(uv*8.+t*.1)*4.)*.5+sin(uv.x*91.-t*.7+uv.y*67.)*.23;vec2 ripple=vec2(wave,sin(uv.x*74.+uv.y*150.-t))*p_ripple*.003*(.2+distance*2.);vec2 pointer=vec2(iCursor.x,1.-iCursor.y);float r=length((uv-pointer)*vec2(iResolution.x/iResolution.y,1.));ripple+=normalize(uv-pointer+vec2(.0001))*sin(r*95.-t*2.)*exp(-r*6.)*p_interaction*iActive*.002;vec3 col=wmBase(uv+ripple*water);vec2 mirrored=vec2(uv.x,2.*p_horizon-uv.y)+ripple;vec3 reflection=wmBase(mirrored);col=mix(col,reflection,water*p_reflection*.55);float sheen=pow(max(0.,wave*.5+.5),12.)*exp(-((uv.x-.65)/(.08+distance*.8))*((uv.x-.65)/(.08+distance*.8)))*water;col+=sheen*p_sheen*vec3(.14,.08,.04);c=vec4(col,1);}
